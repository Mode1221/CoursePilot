# CoursePilot 운영 크론. deploy.sh 가 이 파일을 crontab 에 설치한다.
#   설치 확인: crontab -l
#   {{ROOT}} 는 설치 시 저장소 절대 경로로 치환된다.
#
# 배치는 백엔드 컨테이너 안에서 돈다(파이썬·의존성이 거기 있다).
# nice -n 19 로 낮춘다 — VM 이 1 OCPU 라 배치가 CPU 를 잡으면 API 응답이 밀린다.
# nice 는 **컨테이너 안의 python 앞**에 둔다. docker compose 앞에 두면 CLI 만 낮아지고 배치 프로세스는 그대로다.
# 실행이 겹치면 뒤에 뜬 쪽이 batch_lock 으로 빠지므로 중복은 걱정하지 않아도 된다.
SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin

# 폐업 대장(LOCALDATA) 내려받기 — 매주 월요일 02:00
0 2 * * 1 cd {{ROOT}} && docker compose -f docker-compose.prod.yml exec -T backend nice -n 19 python scripts/fetch_localdata.py >> {{ROOT}}/logs/localdata.log 2>&1

# 상권 수집·보강 — 매일 03:00 (영업시간 160건/일·평점 11건/일로 나눠 채운다)
0 3 * * * cd {{ROOT}} && docker compose -f docker-compose.prod.yml exec -T backend nice -n 19 python scripts/build_places.py >> {{ROOT}}/logs/build_places.log 2>&1

# 저장된 장소 갱신 — 매일 04:30 (폐업 전체 / 영업시간 30일 / 평점 90일 TTL)
30 4 * * * cd {{ROOT}} && docker compose -f docker-compose.prod.yml exec -T backend nice -n 19 python scripts/refresh_places.py >> {{ROOT}}/logs/refresh_places.log 2>&1

# 요즘 뜨는 곳(검색 트렌드·리뷰 증가·비협찬 블로그) + 진행 중인 팝업·전시 — 매일 05:30
30 5 * * * cd {{ROOT}} && docker compose -f docker-compose.prod.yml exec -T backend nice -n 19 python scripts/refresh_hot.py >> {{ROOT}}/logs/refresh_hot.log 2>&1

# 가입하지 않은 체험 계정(30일 지남)·지난 사용 한도 카운터 정리 — 매일 05:45
45 5 * * * cd {{ROOT}} && docker compose -f docker-compose.prod.yml exec -T backend nice -n 19 python scripts/purge_guests.py >> {{ROOT}}/logs/purge_guests.log 2>&1

# DB 백업 — 매일 05:00 (7일 보관, 실패 시 웹훅 알림)
0 5 * * * {{ROOT}}/scripts/ops/backup.sh >> {{ROOT}}/logs/backup.log 2>&1

# 디스크 사용률 점검 — 6시간마다 (85% 초과 시 웹훅 알림)
0 */6 * * * {{ROOT}}/scripts/ops/disk_check.sh >> {{ROOT}}/logs/disk.log 2>&1

# 자동 배포 — 5분마다 main 을 확인해 새 커밋이면 그 sha 이미지로 배포, 실패하면 되돌린다.
# 끄려면 .env 에 AUTO_DEPLOY=false.
*/5 * * * * {{ROOT}}/scripts/ops/auto_deploy.sh >> {{ROOT}}/logs/deploy.log 2>&1
